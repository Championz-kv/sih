 307| async function runAiCategorize(){
 308|   if(sahayaBusy) return;
 309|   sahayaBusy = true;
 310|   aiCancelled = false;
 311|   setSahayaLoading(true);
 312| 
 313|   const title = val('fTitle');
 314|   const description = val('fDesc');
 315|   const district = val('districtSelect');
 316|   const state = val('stateSelect');
 317|   const aff = val('fAffected');
 318|   const affected_count = aff ? parseInt(aff, 10) : null;
 319| 
 320|   try{
 321|     const res = await fetch('/api/categorize', {
 322|       method: 'POST',
 323|       headers: { 'Content-Type': 'application/json' },
 324|       body: JSON.stringify({ title, description, district, state, affected_count })
 325|     });
 326|     if(aiCancelled) return;              /* user cancelled — ignore late response */
 327|     if(!res.ok) throw new Error('HTTP ' + res.status);
 328|     const result = await res.json();
 329|     if(result && result.error) throw new Error(result.error);
 330| 
 331|     applyAiResult(result);
 332|     if(aiCancelled) return;              /* user cancelled while applying */
 333|     setAiVerified();
 334|   }catch(err){
 335|     if(aiCancelled) return;              /* cancelled — no error toast */
 336|     console.error('[sahayak] categorize failed:', err);
 337|     toast('Sahayak could not categorize — please select manually');
 338|     setNeedsReview();
 339|     /* Step 2 is already active — the user can pick categories by hand. */
 340|   }finally{
 341|     sahayaBusy = false;
 342|     if(!aiCancelled) setSahayaLoading(false);   /* don't pop the overlay if user cancelled */
 343|   }
 344| }

 346| function applyAiResult(result){
 347|   /* Categories — match Gemini strings to chips whose text matches exactly. */
 348|   const chips = {};
 349|   document.querySelectorAll('#categoryGroups .cat-tag').forEach(el => { chips[el.textContent.trim()] = el; });
 350| 
 351|   document.querySelectorAll('#categoryGroups .cat-tag').forEach(el => el.classList.remove('sel'));
 352|   selectedTags.clear();
 353|   (Array.isArray(result.categories) ? result.categories : []).slice(0, 4).forEach(cat => {
 354|     const chip = chips[cat];
 355|     if(chip){
 356|       chip.classList.add('sel');
 357|       selectedTags.add(cat);
 358|     }
 359|   });
 360|   syncSelection();   // rebuilds the Selected mirror (refreshes badge once aiPick is known)
 361| 
 362|   /* Severity */
 363|   const sev = SEVERITY_LABELS[String(result.severity || '').toLowerCase()];
 364|   if(sev) document.getElementById('fSev').value = sev;
 365| 
 366|   /* Urgency — number 1-10 → dropdown option */
 367|   const urg = urgencyLabel(result.urgency);
 368|   if(urg) document.getElementById('fUrg').value = urg;
 369| 
 370|   /* Hashtags */
 371|   const fTags = document.getElementById('fTags');
 372|   if(fTags && Array.isArray(result.hashtags)) fTags.value = result.hashtags.join(' ');
 373| 
 374|   /* Reasoning below the badge */
 375|   const noteEl = document.getElementById('aiNote');
 376|   if(noteEl){
 377|     noteEl.className = 'note';
 378|     noteEl.innerHTML = result.reasoning
 379|       ? '<span style="font-size:11.5px; color:var(--text-faint); font-style:italic; margin-top:4px;">Sahayak: ' + result.reasoning + '</span>'
 380|       : '';
 381|   }
 382| 
 383|   /* Remember the AI suggestion so later user changes flip the badge. */
 384|   aiPick = { tags: [...selectedTags], sev: val('fSev'), urg: val('fUrg') };
 385| }

 387| function isAiMatch(){
 388|   if(!aiPick) return false;
 389|   const sameTags = aiPick.tags.length === selectedTags.size &&
 390|                    aiPick.tags.every(t => selectedTags.has(t));
 391|   return sameTags && val('fSev') === aiPick.sev && val('fUrg') === aiPick.urg;
 392| }

 394| function refreshAiBadge(){
 395|   if(!aiPick) return;
 396|   if(isAiMatch()) setAiVerified(); else setNeedsReview();
 397| }

 412| function postFinal(){
 413|   if(!currentUserId){ go('login.html'); return; }
 414|   const postBtn = document.getElementById('postBtn');
 415|   postBtn.disabled = true;
 416|   (async () => {
 417|     try{
 418|       await uploadEvidenceFiles();        /* evidence → evidences bucket */
 419|       const payload = collectFormData();
 420|       /* AI-verified cases post straight through; anything else needs review. */
 421|       payload.problem_state = (aiVerified === true) ? 'reviewed' : 'unreviewed';
 422| 
 423|       const draftId = sessionStorage.getItem('ss_draft_id');
 424|       let saved;
 425|       if(draftId){
 426|         /* if the draft row was deleted meanwhile, submit as a fresh problem */
 427|         saved = await sbClient.from('problems')
 428|           .update(payload).eq('id', draftId).select('case_no').maybeSingle();
 429|         if(!saved.data){
 430|           saved = await sbClient.from('problems')
 431|             .insert(payload).select('case_no').single();

 569| function safeEvidenceName(name){
 570|   const base = String(name).replace(/^.*[\\/]/, '');   /* strip any folder path */
 571|   return base.replace(/[^\w.\-()]+/g, '_').slice(-120);/* url-safe, keep extension */

 573| /* true if `path` already exists in the evidences bucket — never throws,
 574|    never surfaces an error (a blind listing just means "treat as new"). */
 575| async function evidenceFileExists(path){
 576|   try{
 577|     const parts = String(path).split('/');
 578|     const name = parts.pop();
 579|     const folder = parts.join('/');
 580|     const { data, error } = await sbClient.storage.from(EVIDENCE_BUCKET)
 581|       .list(folder, { limit: 1, search: name });
 582|     if(error) return false;
 583|     return (data || []).some(f => f.name === name);
 584|   }catch(e){ return false; }
 585| }
 586| function evidencePublicUrl(path){
 587|   try{
 588|     const { data } = sbClient.storage.from(EVIDENCE_BUCKET).getPublicUrl(path);
 589|     return (data && data.publicUrl) || null;
 590|   }catch(e){ return null; }
 591| }
 592| /* Upload one file → public URL. If it already exists in the bucket, reuse the
 593|    existing object silently (no error shown, no duplicate upload). */
 594| async function uploadEvidenceFile(file){
 595|   if(!currentUserId || !file) return null;
 596|   const path = currentUserId + '/' + safeEvidenceName(file.name);
 597|   try{
 598|     if(await evidenceFileExists(path)) return evidencePublicUrl(path);
 599|     const { error } = await sbClient.storage.from(EVIDENCE_BUCKET)
 600|       .upload(path, file, { upsert: false, contentType: file.type || undefined });
 601|     if(error){
 602|       /* upload raced with another tab or the listing missed it — re-check and
 603|          reuse the existing object silently */
 604|       if(await evidenceFileExists(path)) return evidencePublicUrl(path);
 605|       console.error('[submit] evidence upload failed:', error.message);
 606|       toast('Could not upload ' + file.name + ' — ' + error.message);
 607|       return null;
 608|     }
 609|     return evidencePublicUrl(path);
 610|   }catch(e){
 611|     console.error('[submit] evidence upload failed:', e);
 612|     return null;
 613|   }
 614| }
