function toggleFilterGroup(headerEl){ headerEl.closest('.filter-group').classList.toggle('collapsed'); }
function clearExploreFilters(){
  document.getElementById('exploreSearch').value = '';
  document.querySelectorAll('.filter-box input[type="checkbox"]').forEach(cb => cb.checked = false);
  renderExplore();
}
function getChecked(containerId){
  return Array.from(document.querySelectorAll('#'+containerId+' input[type="checkbox"]:checked')).map(cb => cb.value);
}
function renderExplore(){
  const q = document.getElementById('exploreSearch').value;
  const sortMode = document.getElementById('exploreSort').value;
  const cats = getChecked('filterCategory');
  const sevs = getChecked('filterSeverity');
  const stats = getChecked('filterStatus');

  let list = searchProblems(PROBLEMS, q);
  list = filterProblems(list, { categories:cats, severities:sevs, statuses:stats });
  list = sortProblems(list, sortMode);

  document.getElementById('exploreCount').textContent = `${list.length} matching problem${list.length===1?'':'s'}`;
  document.getElementById('exploreGrid').innerHTML = list.length ? list.map(problemCardHTML).join("")
    : `<div class="empty" style="grid-column:1/-1;"><h4>No problems match those filters</h4><p>Try widening your search or clearing a filter.</p></div>`;
}

document.addEventListener('DOMContentLoaded', () => {
  renderShell();

  const params = new URLSearchParams(location.search);

  /* Category filter — the full posting taxonomy, grouped by heading.
     Problems may carry several sub-tags: a case counts toward every tag it
     carries, and filterProblems() matches any of them. Zero-count tags stay
     visible but dimmed so the whole taxonomy remains discoverable. */
  const tagCounts = {};
  PROBLEMS.forEach(p => problemCats(p).forEach(t => tagCounts[t] = (tagCounts[t]||0)+1));
  const catHTML = PROBLEM_CATEGORY_GROUPS.map(gr => {
    const total = gr.tags.reduce((s,t)=> s+(tagCounts[t]||0), 0);
    const rows = gr.tags.map(t => {
      const n = tagCounts[t] || 0;
      return `<label class="check-row${n ? '' : ' dimmed'}">` +
        `<input type="checkbox" value="${t}">` +
        `<span class="cat-dot" style="background:${gr.color};"></span>${t}` +
        `<span class="n">${n || ''}</span></label>`;
    }).join("");
    return `<div class="filter-cat-group">` +
      `<div class="filter-subhead"><span class="cat-dot" style="background:${gr.color};"></span>${gr.name}<span class="n">${total || ''}</span></div>` +
      rows + `</div>`;
  }).join("");
  document.getElementById('filterCategory').innerHTML =
    '<h5 class="filter-toggle" onclick="toggleFilterGroup(this)">Category <svg class="chev" width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><polyline points="6 9 12 15 18 9"/></svg></h5>' +
    '<div class="filter-body filter-body-cats">' + catHTML + '</div>';

  if(params.get('q')) document.getElementById('exploreSearch').value = params.get('q');

  document.getElementById('exploreSearch').addEventListener('input', renderExplore);
  document.getElementById('exploreSort').addEventListener('change', renderExplore);
  document.querySelectorAll('.filter-box input[type="checkbox"]').forEach(cb => {
    cb.addEventListener('change', renderExplore);
  });

  renderExplore();

  /* Live supporter counts — overlay problems.support from the DB onto the
     demo list, then re-render so every ♥ on the cards matches the detail
     page exactly (same column, no duplicate counter). Demo values stay as
     the offline fallback when Supabase is unreachable. */
  fetchSupportCounts().then(map => {
    if(applySupportCounts(map)) renderExplore();
  });
});
